#-deployment
apiVersion: apps/v1
kind: Deployment
metadata:
  name: postgres-deployment
  namespace: jernery
  labels:
    app.kubernetes.io/name: jerney-db
    app.kubernetes.io/part-of: jerney
    app.kubernetes.io/component: database
spec:
  replicas: 1
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app.kubernetes.io/name: jernery-db
  template:
    metadata:
      labels:
        app.kubernetes.io/name: jernery-db
        app.kubernetes.io/part-of: jernery
        app.kubernetes.io/component: database
    spec:
      automountServiceAccountToken: false
      tolerations:
        - key: "karpenter.sh/disrupted"
          operator: "Exists"
          effect: "NoSchedule"
      containers:
        - name: postgres
          image: postgres:16-alpine 
          ports:
            - containerPort: 5432
              protocol: TCP
          envFrom:
            - secretRef:
                name: jernery-secret
          volumeMounts:
            - name: db_storage
              mountPath: /var/lib/postgresql/data
              subPath: pgdata
          resources:
            requests:
              cpu: 250m
              memory: 256Mi
          limits:
            cpu: 500m
            memory: 512Mi
          securitContext:
            allowPrivilegeEscaltion: false
            readOnlyRootFileSystem: false
            runAsNonRoot: false
          livenessProbe:
            exec:
              command: [ "pg_ready", "-U", "jernery_user", "-d", "jernery_db" ]
            initialDelaySeconds: 15
            periodSeconds: 10
            timeoutSeconds: 5
          readinessProbe:
            exec:
              command: [ "pg_ready", "-U", "jernery_user", "-d", "jernery_db" ]
            initialDelaySeconds: 15
            periodSeconds: 10
            timeoutSeconds: 5
      volumes:
        - name: db_storage
          persistentVolumeClaim:
            claimName: pvc
---

#-backend-deployment
apiVersion: app/v1
kind: Deployment
metdata:
  name: backend_deployment
  namespace: jernery
  labels:
    app.kubernetes.io/name: jernery-backend
    app.kubernetes.io/part-of: jernery
    app.kubernetes.io/component: backend
spec:
  replicas: 2
  selector:
    matchLabels:
      app.kubernetes.io/name: jernery-backend
  template:
    metadata:
      labels:
        app.kubernetes.io/name: jernery-backend
        app.kubernetes.io/part-of: jernery
        app.kubernetes.io/component: backend
    spec:
      automountServiceAccount: false
      toleration:
        - key: "karpenter.sh/disrupted"
          operator: "Exists"
          effect: "NoSchedule"
      initContainers:
        - name: wait-for-db
          image: busybox:1.36
          command:
            - sh
            - c
            - |
              echo "waiting for jernery-db:5432..."
              until nc -z jernery-db 5432; do
                echo "DB not ready, retrying in 3s ...."
                sleep 3
              done
              echo "DB is ready"
      containers:
        - name: backend
          image: 
            ports:
              - containerPorts: 5000
                protocol: TCP
            env:
              - name: PORT
                value: "5000"
              - name: DB_HOST
                value: "jernery_db"
              - name: DB_PORT
                value: "5432"
              - name: DB_USER
                valueFrom:
                  secretKeyRef:
                    name: secret 
                    key: POSTGRES_USER
              - name: DB_PASSWORD
                valueFrom:
                  secretKeyRef:
                    name: secret
                    key: POSTGRES_PASSWORD
              - name: DB_NAME
                valueFrom:
                  secretKeyRef:
                    name: secret
                    key: POSTGRES_DB
          resources:
            requests:
              cpu: 100m
              memory: 128Mi
            limits:
              cpu: 300m
              memory: 256Mi
          securityContext:
            allowPrivilegeEscalation: false
            readonlyRootFilesystem: true
            runAsNonRoot: true
            runAsUser: 1000
            capabilities:
              drop:
                - ALL
            livenessProbe:
              httpGet:
                path: /api/health
                port: 5000
              initialDelaySeconds: 10
              periodSeconds: 15
              timeoutSeconds: 5
            readinessProbe:
              httpGet:
                path: /api/health
                port: 5000
              initialDeloySeconds:  5
              periodSeconds: 5
              timeoutseconds: 3
